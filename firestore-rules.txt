// انسخ الكود ده والصقه في:
// Firebase Console -> Firestore Database -> Rules -> استبدل الموجود -> Publish

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // أي حد بيزور الموقع العام يقدر "يقرأ" بس
    // الكتابة/التعديل/الحذف مسموح فقط لو داخل بحساب الأدمن (Authenticated)
    match /{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
